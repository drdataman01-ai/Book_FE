\begin{FEinfographic}{ch:10}{do correlations go to one in a crash, so that diversification breaks down when it is needed most?}

\infopanel{1}{Three things to know about crisis correlations}{%
\infocards
  {Among stocks}{Correlations of large losses rise in bear markets, across countries and industries.}
  {The volatility bias}{Correlations look higher whenever volatility is high, so part of the rise is a statistical effect.}
  {Stocks and bonds}{Bonds protected stocks from 2000 to 2021, but not when inflation dominated, as in 2022.}}

\infopanel{2}{Where the evidence stands: what published studies found}{%
\begin{infoscore}
\scoreitem{y}{Yes}{Stock correlations in bear markets}
\scoreitem{p}{Partly}{After correcting for higher volatility}
\scoreitem{y}{Yes}{Other risky assets in U.S. sell-offs}
\scoreitem{n}{No}{Treasury bonds, 2000--2021}
\scoreitem{y}{Yes}{Treasury bonds when inflation dominates}
\scoreitem{n}{Mostly no}{Stock--bond portfolios in the worst months}
\end{infoscore}}

\infopanel{3}{Key numbers from 2022}{%
\infonumbers
  {18.1\%}{fall of the S\&P 500}
  {13\%}{approximate fall of the U.S. dollar bond market}
  {11\%}{loss of a 60/40 portfolio over the year}
  {19\%}{the 60/40 portfolio's deepest fall}}

\infopanel{4}{Lab: Correlations in calm markets and in crises}{%
\begin{infolab}
\item \textbf{A.} Correlations in calm, volatile and worst months, before and after the Forbes--Rigobon correction.
\item \textbf{B.} Tail dependence against a normal benchmark. \textbf{C.} The stock--bond correlation by decade and inflation regime.

\infosummary
  {Correlations among stock markets and industries rise in falling markets by more than the volatility bias explains, but diversification across asset classes has usually held up, and the protection that bonds give depends on inflation.}
  {Plan with crisis correlations, diversify across risks rather than labels, and know whether your protection guards against recessions or inflation shocks.}
