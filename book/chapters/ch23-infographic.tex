\begin{FEinfographic}{ch:23}{foundation models trained or fine-tuned on market data forecast volatility better than GARCH and HAR}

\infopanel{1}{Three routes to a financial foundation model}{%
\infocards
  {Pretrain on markets}{Kronos is trained on more than 12 billion price bars from 45 exchanges.}
  {Fine-tune}{A general model such as TimesFM is adapted to volatility data, using only past information.}
  {Generate paths}{Synthetic price paths can widen a stress test beyond history's few crises.}}

\infopanel{2}{Evidence scoreboard}{%
\begin{infoscore}
\scoreitem{p}{Partly}{Kronos, volatility}
\scoreitem{p}{Partly}{Kronos, synthetic data}
\scoreitem{y}{Yes}{Fine-tuned TimesFM}
\scoreitem{p}{Partly}{Zero-shot TimesFM}
\scoreitem{n}{No}{Direction of volatility}
\scoreitem{p}{Weakly}{Against specialized models}
\scoreitem{o}{Untested}{Clean post-release data}
\end{infoscore}}

\infopanel{3}{Key numbers}{%
\infonumbers
  {9\%}{lower error for Kronos on its authors' benchmarks}
  {21}{indices on which fine-tuned TimesFM beat GARCH and HAR}
  {512}{days of history for the best zero-shot TimesFM}
  {82.69}{VIX close on March 16, 2020, a record}}

\infopanel{4}{Lab: replicating Kronos and fine-tuned TimesFM on daily market data}{%
\begin{infolab}
\item \textbf{A.} Zero-shot return and variance forecasts. \textbf{B.} Fine-tuned TimesFM against GARCH and HAR.
\item \textbf{C.} History length and fine-tuning choices. \textbf{D.} Synthetic paths from Kronos and a $t$-GARCH.
\item \textbf{E.} 99\% VaR backtests. \textbf{F.} The same tests on days after each model's release only.
\end{infolab}}

\infoverdict
  {Training on market data appears to help, and a general model fine-tuned on volatility data can match and sometimes beat the best econometric models. But the evidence is young and mostly from working papers, and none of it yet uses data the models could not have seen.}
  {Keep HAR and GARCH as the yardstick, and admit a foundation model into a risk system only if it beats them out of sample by enough to change a risk decision.}
\end{FEinfographic}
