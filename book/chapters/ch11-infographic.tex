\begin{FEinfographic}{ch:11}{high-interest currencies fall enough to cancel their extra yield, so the carry trade earns nothing on average}

\infopanel{1}{The carry trade in three ideas}{%
\infocards
  {The trade}{Borrow in low-interest currencies such as the yen and invest in high-interest ones.}
  {The theory}{Uncovered interest parity says the high-yield currency should fall by its interest advantage.}
  {The catch}{Carry earns slowly in calm years and crashes in global panics: up the stairs, down the elevator.}}

\infopanel{2}{Evidence scoreboard}{%
\begin{infoscore}
\scoreitem{n}{No}{Fama slope equal to 1}
\scoreitem{n}{No}{Carry portfolios earn nothing}
\scoreitem{y}{Yes}{Carry loses in global downturns}
\scoreitem{n}{No}{Rare disasters not yet seen explain it}
\scoreitem{p}{Partly}{Long horizons and very high inflation}
\scoreitem{o}{Unclear}{Carry after 2008}
\end{infoscore}}

\infopanel{3}{Key numbers}{%
\infonumbers
  {$-0.88$}{average Fama slope across 75 estimates}
  {9}{developed currencies in the lab}
  {over 1/3}{fall of the Australian dollar against the yen, 2008}
  {12.4\%}{Nikkei fall on August 5, 2024}}

\infopanel{4}{Lab: replicating Fama (1984) and Lustig, Roussanov and Verdelhan (2011)}{%
\begin{infolab}
\item \textbf{A.} Fama regressions of exchange-rate changes on interest gaps, with Newey--West errors.
\item \textbf{B.} A carry portfolio split into interest earned and currency change, and its distribution.
\item \textbf{C.} Subperiods, costs, exposure to stocks and the VIX. \textbf{D.} Drawdowns and recovery times.
\end{infolab}}

\infoverdict
  {High-interest currencies have not fallen by their interest advantage on average, so the carry trade has earned a premium, but that premium is paid for with sudden crashes in global downturns.}
  {Measure carry's correlation with stocks in bad months, not on average, and ask how much leverage a carry fund uses and how it did in late 2008 and August 2024.}
