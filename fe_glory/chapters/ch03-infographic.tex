\begin{FEinfographic}{ch:03}{size and value explain returns that beta cannot, but testing many factors produces false discoveries}

\infopanel{1}{From one factor to the factor zoo}{%
\infocards
  {Beta}{The CAPM prices risk by exposure to the market alone, and average returns rise with beta less steeply than it predicts.}
  {Size and value}{Adding SMB and HML explains most of the differences across the 25 size and book-to-market portfolios.}
  {The zoo}{Hundreds of factors tested on the same data make many false discoveries, so a new factor needs a higher bar.}}

\infopanel{2}{Evidence scoreboard}{%
\begin{infoscore}
\scoreitem{n}{No}{Beta alone explains average returns}
\scoreitem{y}{Yes}{Size and value across the 25 portfolios}
\scoreitem{p}{Partly}{Value alongside profitability and investment}
\scoreitem{o}{Disputed}{Why size and value earn a premium}
\scoreitem{y}{Yes}{Apparent fund-manager skill is factor exposure}
\scoreitem{n}{No}{Hundreds of new factors judged at $t > 2$}
\end{infoscore}}

\infopanel{3}{Key numbers}{%
\infonumbers
  {90\%+}{$R^2$ for most of the 25 portfolios with three factors}
  {about 50}{of 1,000 zero-premium factors look significant at 1.96}
  {$t \approx 3$}{the safer hurdle for a new factor}
  {1963}{start of the monthly sample of Fama and French}}

\infopanel{4}{Lab: replicating Fama and French (1993) and the factor zoo}{%
\begin{infolab}
\item \textbf{A.} The CAPM and the three-factor model on the 25 size and book-to-market portfolios: alphas, loadings and $R^2$.
\item \textbf{B.} 1,000 simulated factors with no true premium, tested at $|t| > 1.96$ and at $|t| > 3$.
\item \textbf{C.} Annualized premiums and Sharpe ratios, and whether each premium survives trading costs.
\end{infolab}}

\infoverdict
  {Adding size and value to the market explains differences in portfolio returns that beta alone cannot, but many of the factors proposed since are likely to be false discoveries, so a new factor must clear a higher bar.}
  {Separate alpha from factor exposure: cheap index and factor funds already provide size and value, so check how much of an active fund's return they explain.}
\end{FEinfographic}
